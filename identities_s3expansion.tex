\section{Action Ansatz}
\begin{equation} 
S = S_2 + S_3 
 \end{equation}
 where
 \begin{equation} 
\begin{aligned}S_2 = \int\frac{d^D p}{(2\pi)^D}\Bigl[
    \p^{\mu\nu}(-p)\,K^{\phi\phi}_{\mu\nu\rho\sigma}(p)\p_{\rho\sigma}(p) + \p^{\mu\nu}(-p)K^{\phi G}_{\mu\nu\lambda}\,^{\rho\sigma}(p) \G^\lambda _{\rho\sigma}(p)+ \G^\lambda _{\rho\sigma}(-p)K^{G G}_{\lambda}\,^{\mu\nu}\,_{\rho}\,^{\sigma\tau}(p) \G^\rho _{\sigma\tau}(p) \Bigr]
\end{aligned}\end{equation}
\hfill \break
\begin{equation} \begin{aligned} 
S_3 &= S_{\phi\phi\phi} + S_{\phi\phi G} + S_{\phi G G} \\
S_{\phi\phi\phi}  &= \int \pmeas \pmeas[q] \p^{\mu\nu} (-p) \p^{\alpha \beta} (-q) \p^{\rho\sigma} (p+q) K^{\phi\phi\phi}_{\mu\nu\alpha\beta\rho\sigma} (p,q) \\
S_{\phi\phi G} &= \int \pmeas \pmeas[q] \p^{\mu\nu} (-p) \p^{\alpha\beta}(-q) K^{\phi\phi G}_{\mu\nu\alpha\beta\lambda}\,^{\rho\sigma} (p,q) \G^\lambda _{\rho\sigma} (p+q)\\
S_{\phi G G}&= \int \pmeas \pmeas[q] \p^{\mu\nu} (-p-q) K^{\phi G G}_{\mu\nu\lambda}\,^{\alpha\beta}\,_{\rho}\,^{\sigma\tau} \G^\lambda _{\alpha\beta} (p) \G^\rho _{\sigma \tau} (q)
 \end{aligned} \end{equation}



\section{Gauge transformations}
\begin{equation} 
\begin{aligned} 
\delta \p^{\mu\nu} = \delta^{(0)}\p^{\mu\nu} + \delta^{(1)}\p^{\mu\nu}(p) &= D_{\phi^0}^{\mu\nu}\,_{\alpha} (p)\tomega^\alpha (p) + \int \pmeas[d]{q} \p^{\alpha\beta}(p-q)\tomega^\rho (q) D_{\phi^1}^{\mu\nu}\,_{\alpha\beta\rho}(p,q) \\ 
D_{\phi^0}^{\mu\nu}\,_{\alpha} (p) &= -i(\eta^{\mu\nu}p_\alpha - p^{(\mu}\delta^{\nu)}_{\alpha}) \\  
D_{\phi^1}^{\mu\nu}\,_{\alpha\beta\rho}(p,q) &= -i (\delta^{\mu}_\alpha \delta^\nu _\beta p_\rho - q_\beta \delta^{(\mu}_\alpha \delta^{\nu)}_\rho)
\end{aligned} 
\end{equation}

\begin{equation} 
\begin{aligned} 
\delta \G^\lambda _{\mu\nu} (p) = \delta^{(0)} \G^\lambda _{\mu\nu} (p) + \delta^{(1)} \G^\lambda _{\mu\nu} (p) &=D_{G^0}^{\lambda}\,_{\mu\nu\alpha}(p) \tomega^\alpha (p) + \int \pmeas[d]{q} \G^{\alpha}_{\beta\gamma} (p-q) \tomega^\rho (q) D_{G^1}^{\lambda}\,_{\mu\nu\alpha}\,^{\beta\gamma}\,_{\rho} (p,q) \\ 
D_{G^0}^{\lambda}\,_{\mu\nu\alpha}(p) &= -p_\mu p_\nu \delta^{\lambda}_{\alpha} +\frac{1}{2} \delta^{\lambda}_{(\mu}p_{\nu)} p_{\alpha} \\  
D_{G^1}^{\lambda}\,_{\mu\nu\alpha}\,^{\beta\gamma}\,_{\rho} (p,q) &= -i[\delta^\lambda _\alpha \delta^\beta _\mu \delta^\gamma _\nu (p-q)_\rho - q_\alpha \delta^\beta _\mu \delta^\gamma _\nu \delta^\lambda _\rho + \delta^\lambda _\alpha \delta^\beta _\rho \delta^\gamma _{(\mu}q_{\nu)}]
\end{aligned} 
\end{equation}
